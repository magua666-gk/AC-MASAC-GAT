# Experiment commands

# 1. Main experiment: MASAC + GAT attention + curriculum learning
python main_SAC_curriculum.py --experiment_type curriculum --use_curriculum

# Optional: adaptive alpha instead of fixed alpha=1.0
python main_SAC_curriculum.py --experiment_type curriculum --use_curriculum --adaptive_alpha --target_entropy -0.1 --alpha_max 1.0

# 2. Ablation: no attention, with curriculum learning
# This uses the current main pipeline and only replaces GAT with simple MLP aggregation.
python main_SAC_curriculum.py --experiment_type no_attention

# 3. Ablation: GAT attention, no curriculum learning
# This uses the current main pipeline and trains one fixed task for a fixed number of episodes.
python main_SAC_curriculum.py --experiment_type no_curriculum --fixed_episodes 800

# 4. H-CRRT traditional baseline test
python main_SAC_curriculum.py --experiment_type h_crrt --test_episodes 100 --no_render

# Faster H-CRRT smoke/preview run
python main_SAC_curriculum.py --experiment_type h_crrt --test_episodes 20 --no_render --rrt_max_nodes 500 --shortcut_iters 50

# Higher-quality H-CRRT baseline run
python main_SAC_curriculum.py --experiment_type h_crrt --test_episodes 100 --no_render --rrt_max_nodes 1500 --shortcut_iters 200 --replan_horizon 50


# Test commands

# Main experiment single-difficulty test.
# If --model_path is omitted, the latest model under outputs/train/curriculum/ is selected automatically.
python main_SAC_curriculum.py --experiment_type curriculum --test --no_render --test_episodes 100 --hero_count 1 --enemy_count 3  --obstacle_count 2

# Main experiment multi-difficulty test
python main_SAC_curriculum.py --experiment_type curriculum --multi_difficulty_test --no_render --test_episodes 100

# No-attention test. If --model_path is omitted, the latest no_attention model is selected automatically.
python main_SAC_curriculum.py --experiment_type no_attention --test --no_render --test_episodes 100

# No-curriculum test. If --model_path is omitted, the latest no_curriculum model is selected automatically.
python main_SAC_curriculum.py --experiment_type no_curriculum --test --no_render --test_episodes 100

# Explicit model path example
python main_SAC_curriculum.py --experiment_type curriculum --test --no_render --model_path outputs/train/curriculum/<run_dir>/training_runs/<train_run>/models/final/final_model_<timestamp>
