python main_SAC_curriculum.py --experiment_type no_curriculum --test --no_render --test_episodes 100

# Leader predictive safety shield sensitivity test.
# This fixes the trained policy and sweeps shield-off, default, safe distance, warning distance, and prediction horizon.
python main_SAC_curriculum.py --experiment_type curriculum --safety_sensitivity_test --no_render --test_episodes 100 --hero_count 1 --enemy_count 3 --obstacle_count 2

python main_SAC_curriculum.py --experiment_type curriculum --safety_sensitivity_test --no_render --test_episodes 100 --hero_count 1 --enemy_count 3 --obstacle_count 2 --safe_distance_values 25,35,45,55,65 --warning_distance_values 80,120,150,180,220 --horizon_values 2,4,6,8,10

# Single test with Leader safety shield disabled.
python main_SAC_curriculum.py --experiment_type curriculum --test --no_render --test_episodes 100 --hero_count 1 --enemy_count 3 --obstacle_count 2 --disable_leader_safety_shield
