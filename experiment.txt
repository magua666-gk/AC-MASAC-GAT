# Training commands

# 1. Main experiment: MASAC + attention + curriculum learning
python main_SAC_curriculum.py --experiment_type curriculum --use_curriculum

# Optional: enable adaptive alpha instead of the default fixed alpha=1
python main_SAC_curriculum.py --experiment_type curriculum --use_curriculum --adaptive_alpha

# 2. Standard MASAC: no curriculum learning
python main_SAC_curriculum.py --experiment_type standard

# Standard MASAC with adaptive alpha
python main_SAC_curriculum.py --experiment_type standard --adaptive_alpha

# 3. Ablation: no attention, with curriculum learning
python main_SAC_curriculum.py --experiment_type no_attention

# 4. Ablation: attention, no curriculum learning
python main_SAC_curriculum.py --experiment_type no_curriculum

# Test commands

# 5. H-CRRT baseline test
python main_SAC_curriculum.py --experiment_type h_crrt --test_episodes 100 --no_render

# Main experiment single-difficulty test.
# If --model_path is omitted, the latest model under outputs/train/curriculum/ is selected automatically.
python main_SAC_curriculum.py --experiment_type curriculum --test

# Main experiment multi-difficulty test
python main_SAC_curriculum.py --experiment_type curriculum --multi_difficulty_test

# Main experiment test with an explicit timestamped model file
python main_SAC_curriculum.py --experiment_type curriculum --test --model_path outputs/train/curriculum/<run_dir>/model/final/final_model_<timestamp>

# no_attention test
python main_SAC_curriculum.py --experiment_type no_attention --test --model_path outputs/train/no_attention/<run_dir>/model/final/final_model_<timestamp>

# no_curriculum test
python main_SAC_curriculum.py --experiment_type no_curriculum --test --model_path outputs/train/no_curriculum/<run_dir>/model/final/final_model_<timestamp>

# 固定 alpha=1，对照组
python main_SAC_curriculum.py --use_curriculum

# 论文式自适应 alpha，但限制最大值
python main_SAC_curriculum.py --use_curriculum --adaptive_alpha --target_entropy -0.1 --alpha_max 1.0

# 更稳定的自适应 alpha
python main_SAC_curriculum.py --use_curriculum --adaptive_alpha --target_entropy -2.0 --alpha_max 1.0
